## 2. 状态与数据模型（`src-tauri/src/state.rs`）

```rust
use serde::{Serialize, Deserialize};
use std::sync::Mutex;

/// 用户状态：空闲 / 工作中 / 休息中
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum UserState { Idle, Working, Resting }

/// 单个工作/休息片段
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Segment {
    pub start_ms: u64,
    pub end_ms: u64,
    pub duration_sec: u64,
}

/// 今日数据（按日落盘）
#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct DailyData {
    pub date: String,
    pub work_segments: Vec<Segment>,
    pub rest_segments: Vec<Segment>,
    pub water_intakes: Vec<String>,   // "09:15"
    pub rest_count: u32,
    pub rest_reminders: u32,
    pub last_activity_ms: u64,
    pub last_water_prompt_ms: u64,
}

/// 用户设置（独立文件）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Settings {
    pub work_threshold_min: u64,   // 默认 45
    pub rest_threshold_min: u64,   // 默认 5
    pub reminder_mode: String,     // "toast" | "popup" | "fullscreen"
    pub water_enabled: bool,       // 默认 true
    pub water_interval_min: u64,   // 默认 60
    pub work_hours_only: bool,     // 默认 false
    pub work_start: String,        // "09:00"
    pub work_end: String,          // "18:00"
    pub paused: bool,              // 默认 false
}

impl Default for Settings {
    fn default() -> Self {
        Settings {
            work_threshold_min: 45, rest_threshold_min: 5,
            reminder_mode: "toast".into(), water_enabled: true,
            water_interval_min: 60, work_hours_only: false,
            work_start: "09:00".into(), work_end: "18:00".into(),
            paused: false,
        }
    }
}

/// 全局共享状态
pub struct AppState {
    pub user_state: UserState,
    pub daily: DailyData,
    pub settings: Settings,
    pub segment_start_ms: u64,        // 当前片段起点
    pub rest_fired_for_segment: bool, // 本片段是否已提醒
    pub snooze_until_ms: u64,         // 稍后提醒截止时间
}

impl AppState {
    pub fn new(settings: Settings, daily: DailyData) -> Self {
        AppState {
            user_state: UserState::Idle,
            daily, settings,
            segment_start_ms: 0, rest_fired_for_segment: false,
            snooze_until_ms: 0,
        }
    }

    /// 每秒 tick：根据最近活动时间推进状态机
    pub fn tick(&mut self, now_ms: u64) {
        if self.settings.paused { return; }
        let rest_th = self.settings.rest_threshold_min * 60_000;
        let since = now_ms.saturating_sub(self.daily.last_activity_ms);

        match self.user_state {
            UserState::Idle => { /* 等待首次活动，由 on_activity 触发 */ }
            UserState::Working => {
                if since >= rest_th {
                    // 闭合工作片段，进入休息
                    self.close_segment(now_ms);
                    self.user_state = UserState::Resting;
                    self.daily.rest_count += 1;
                    // TODO: 开启 rest_segments 记录
                }
            }
            UserState::Resting => {
                if since < rest_th {
                    // 恢复活动，开始新工作片段
                    self.user_state = UserState::Working;
                    self.segment_start_ms = now_ms;
                    self.rest_fired_for_segment = false;
                }
            }
        }
    }

    /// 记录一次活动（由 rdev 回调调用）
    pub fn on_activity(&mut self, now_ms: u64) {
        if self.settings.paused { return; }
        if self.user_state == UserState::Idle {
            self.user_state = UserState::Working;
            self.segment_start_ms = now_ms;
        }
        self.daily.last_activity_ms = now_ms;
    }

    fn close_segment(&mut self, now_ms: u64) {
        let dur = (now_ms.saturating_sub(self.segment_start_ms)) / 1000;
        self.daily.work_segments.push(Segment {
            start_ms: self.segment_start_ms, end_ms: now_ms, duration_sec: dur,
        });
    }

    /// 当前工作片段已连续时长（毫秒）
    pub fn current_segment_ms(&self, now_ms: u64) -> u64 {
        if self.user_state == UserState::Working {
            now_ms.saturating_sub(self.segment_start_ms)
        } else { 0 }
    }
}

/// 供 Tauri 托管的包装
pub struct Store(pub Mutex<AppState>);
```

## 3. 全局活动监听（`src-tauri/src/activity.rs`）

```rust
use crate::state::Store;
use rdev::{listen, Event};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};

/// 启动全局键鼠监听（独立线程，不阻塞主线程）
/// 隐私：rdev 仅回调事件类型/键码，本函数只更新时间戳，绝不记录内容
pub fn start_listener(store: Arc<Store>) {
    std::thread::spawn(move || {
        // 注意：listen 会阻塞当前线程
        if let Err(e) = listen(move |_event: Event| {
            // _event 包含按键类型/坐标，但我们只取“发生了活动”这一事实
            let now = now_ms();
            if let Ok(mut s) = store.0.lock() {
                s.on_activity(now);
            }
        }) {
            eprintln!("[activity] 监听失败: {:?}", e);
        }
    });
}

pub fn now_ms() -> u64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
        .as_millis() as u64
}
```

## 4. 提醒调度（`src-tauri/src/scheduler.rs`）

```rust
use crate::state::{Store, UserState};
use crate::activity::now_ms;
use tauri::Emitter;
use std::sync::Arc;
use std::time::Duration;

/// 启动调度循环：每秒检查阈值，触发休息/喝水提醒事件
pub fn run_scheduler(app: tauri::AppHandle, store: Arc<Store>) {
    std::thread::spawn(move || {
        loop {
            std::thread::sleep(Duration::from_secs(1));
            let now = now_ms();
            let mut s = match store.0.lock() { Ok(g) => g, Err(_) => continue };
            s.tick(now);

            // 休息提醒：连续工作达阈值且本片段未提醒、未处于稍后期
            let work_th = s.settings.work_threshold_min * 60_000;
            let reached = s.current_segment_ms(now) >= work_th;
            let in_work = s.user_state == UserState::Working;
            let not_snoozed = now >= s.snooze_until_ms;

            if in_work && reached && !s.rest_fired_for_segment && not_snoozed {
                s.rest_fired_for_segment = true;
                s.daily.rest_reminders += 1;
                let _ = app.emit("rest_reminder", serde_json::json!({
                    "worked_min": work_th / 60000
                }));
            }

            // 喝水提醒：开启且活跃，且距上次提示超过间隔
            if s.settings.water_enabled
                && in_work
                && now - s.daily.last_water_prompt_ms >= s.settings.water_interval_min * 60_000
            {
                s.daily.last_water_prompt_ms = now;
                let _ = app.emit("water_reminder", ());
            }
            // 锁在作用域结束自动释放
        }
    });
}
```

## 5. 本地存储（`src-tauri/src/storage.rs`）

```rust
use crate::state::{DailyData, Settings};
use std::path::PathBuf;
use std::fs;

fn app_dir() -> PathBuf {
    let mut p = dirs_data();   // 等价于 %APPDATA%/rest-reminder
    p.push("rest-reminder");
    fs::create_dir_all(&p).ok();
    p
}

fn daily_path(date: &str) -> PathBuf {
    let mut p = app_dir();
    p.push("data");
    fs::create_dir_all(&p).ok();
    p.push(format!("{}.json", date));
    p
}

fn settings_path() -> PathBuf {
    let mut p = app_dir();
    p.push("settings.json");
    p
}

pub fn load_settings() -> Settings {
    match fs::read_to_string(settings_path()) {
        Ok(s) => serde_json::from_str(&s).unwrap_or_default(),
        Err(_) => Settings::default(),
    }
}

pub fn save_settings(s: &Settings) {
    if let Ok(j) = serde_json::to_string_pretty(s) {
        fs::write(settings_path(), j).ok();
    }
}

pub fn load_daily(date: &str) -> DailyData {
    match fs::read_to_string(daily_path(date)) {
        Ok(s) => serde_json::from_str(&s).unwrap_or_default(),
        Err(_) => DailyData { date: date.into(), ..Default::default() },
    }
}

pub fn save_daily(d: &DailyData) {
    if let Ok(j) = serde_json::to_string_pretty(d) {
        fs::write(daily_path(&d.date), j).ok();
    }
}

// 简化：真实环境用 dirs  crate 取 %APPDATA%
fn dirs_data() -> PathBuf {
    // Windows: C:\Users\<user>\AppData\Roaming
    PathBuf::from(std::env::var("APPDATA").unwrap_or_else(|_| ".".into()))
}
```

## 6. 系统托盘（`src-tauri/src/tray.rs`）

```rust
use tauri::tray::{TrayIconBuilder, TrayIconEvent};
use tauri::menu::{Menu, MenuItem, MenuEvent};
use tauri::{App, Manager};

pub fn build_tray(app: &App) -> tauri::Result<()> {
    let toggle = MenuItem::with_id(app, "toggle", "暂停检测", true, None::<&str>)?;
    let settings = MenuItem::with_id(app, "settings", "打开设置", true, None::<&str>)?;
    let quit = MenuItem::with_id(app, "quit", "退出应用", true, None::<&str>)?;
    let menu = Menu::with_items(app, &[&toggle, &settings, &quit])?;

    let _tray = TrayIconBuilder::with_id("main-tray")
        .icon(app.default_window_icon().unwrap().clone())
        .menu(&menu)
        .on_menu_event(|app, event: MenuEvent| match event.id().as_ref() {
            "toggle" => { /* invoke command 切换 paused */ }
            "settings" => { if let Some(w) = app.get_window("main") { w.show().ok(); } }
            "quit" => { app.exit(0); }
            _ => {}
        })
        .on_tray_icon_event(|tray, event| {
            if let TrayIconEvent::Click { .. } = event {
                if let Some(w) = tray.app_handle().get_window("main") {
                    w.show().ok(); w.set_focus().ok();
                }
            }
        })
        .build(app)?;
    Ok(())
}
```

## 8. 入口（`src-tauri/src/main.rs`）

```rust
mod activity;
mod state;
mod scheduler;
mod storage;
mod tray;
mod commands;

use state::{AppState, Store, Settings, DailyData};
use std::sync::Arc;
use tauri::Manager;

fn main() {
    let settings: Settings = storage::load_settings();
    let today = chrono::Local::now().format("%Y-%m-%d").to_string();
    let daily: DailyData = storage::load_daily(&today);

    let store = Arc::new(Store(Mutex::new(AppState::new(settings, daily))));
    let store_clone = Arc::clone(&store);

    tauri::Builder::default()
        .setup(move |app| {
            tray::build_tray(app)?;
            activity::start_listener(Arc::clone(&store_clone));
            scheduler::run_scheduler(app.handle().clone(), Arc::clone(&store_clone));
            Ok(())
        })
        .manage(store)
        .invoke_handler(tauri::generate_handler![
            commands::get_settings,
            commands::save_settings,
            commands::get_daily,
            commands::skip_rest,
            commands::toggle_pause,
            commands::record_water,
        ])
        .run(tauri::generate_context!())
        .expect("启动失败");
}
```

## 9. 前端状态管理（`src/stores/appStore.ts`）

```ts
import { defineStore } from 'pinia'
import { invoke } from '@tauri-apps/api/core'
import { listen } from '@tauri-apps/api/event'

export const useAppStore = defineStore('app', {
  state: () => ({
    settings: {} as any,
    daily: {} as any,
    restReminder: null as any,
    waterReminder: false,
  }),
  actions: {
    async init() {
      this.settings = await invoke('get_settings')
      this.daily = await invoke('get_daily')
      // 监听 Rust 端事件，触发对应 UI
      listen('rest_reminder', (e: any) => { this.restReminder = e.payload })
      listen('water_reminder', () => { this.waterReminder = true })
    },
    async saveSettings(s: any) {
      await invoke('save_settings', { settings: s })
      this.settings = s
    },
    async skipRest() { await invoke('skip_rest'); this.restReminder = null },
    async recordWater() {
      await invoke('record_water'); this.waterReminder = false
      this.daily = await invoke('get_daily')
    },
  },
})
```

## 10. 关键隐私与安全工作点

- `activity.rs` 的回调签名中 `_event` 未被读取内容，仅取时间 → 从机制上保证「不记录按键」。
- 所有存储文件不含 PII，且为明文 JSON，用户可审计/删除。
- `settings.paused` 一处开关控制全局监听是否生效（托盘可切换）。
- 提醒事件通过 Tauri 事件总线传递，前端只负责展示，Rust 持有计时真相（避免 WebView 休眠误计）。
